<?php

namespace App\Imports;

use App\Models\Item;
use App\Models\ItemVariation;
use App\Models\Unit;
use App\Models\Warehouse;
use Maatwebsite\Excel\Concerns\ToModel;

class ItemsImport implements ToModel
{
    private $firstRowSkipped = false;
    private $hasErrors = false;
    private $warehouseId;
    protected $rowCount = 0;
    protected $errors = []; // Property to store row errors

    public function __construct($warehouseId)
    {
        $this->warehouseId = $warehouseId;
    }

    public function model(array $row)
    {
        $this->rowCount++;

        if (!$this->firstRowSkipped) {
            $this->firstRowSkipped = true;
            return null;
        }

        if ($this->hasErrors) {
            return null;
        }

        try {
            $warehouse = Warehouse::where('name', $row[0])->first();

            if (!$warehouse || $warehouse->id != $this->warehouseId) {
                throw new \Exception("Row {$this->rowCount}: Warehouse does not match or is missing.");
            }

            $quantity = (!empty($row[12]) && !empty($row[9])) ? (float) $row[9] * (float) $row[12] : $row[9];
            $item = Item::where('item_name', $row[1])
                ->where('warehouse_id', $warehouse->id)
                ->first();

            if (!$item) {
                $item = new Item([
                    'warehouse_id' => $warehouse->id,
                    'item_name' => $row[1],
                    'descriptions' => $row[3],
                    'category' => $row[5],
                    'market' => $row[6],
                    'parent_id' => $row[26],
                ]);
                $item->save();
            }

            $variation = ItemVariation::where('item_id', $item->id)
                ->where('descriptions', $row[7])
                ->where('product_code', $row[8])
                ->where('expired_date', $row[4])
                ->first();

            if ($variation) {
                $variation->quantity += $quantity;
                $variation->reorder_level_stock = $row[10];
                $variation->unit1 = $row[11];
                $variation->unit2 = $row[12];
                $variation->unit3 = $row[13];
                $variation->save();
            } else {
                $variation = new ItemVariation([
                    'item_id' => $item->id,
                    'variations_barcode' => $row[2],
                    'expired_date' => $row[4],
                    'descriptions' => $row[7],
                    'product_code' => $row[8],
                    'quantity' => $quantity,
                    'reorder_level_stock' => $row[10],
                    'unit1' => $row[11],
                    'unit2' => $row[12],
                    'unit3' => $row[13],
                ]);
                $variation->save();
            }
        } catch (\Exception $e) {
            // Store the error with the row count
            $this->errors[] = "Row {$this->rowCount} error: " . $e->getMessage();
        }
    }

    // Method to retrieve errors
    public function getErrors()
    {
        return $this->errors;
    }
}
